-- USDM volume exposure by counterparty category (30d)
SELECT cp.category,
       SUM(t.amount_usd)                              AS volume_usd,
       SUM(t.amount_usd) / SUM(SUM(t.amount_usd)) OVER () AS pct
FROM   transfers_clustered t
JOIN   cross_chain.clusters cp
       ON t.counterparty_cluster = cp.cluster_id
WHERE  t.asset = 'USDM'
  AND  t.block_timestamp >= now() - INTERVAL 30 DAY
GROUP  BY cp.category
ORDER  BY volume_usd DESC;

View Data Solutions query
-- USDM cross-chain bridge & swap flows (last 24h)
SELECT cp.entity_name AS venue,
       cp.category    AS venue_type,
       SUM(CASE WHEN t.direction = 'in'  THEN t.amount_usd END) AS in_24h,
       SUM(CASE WHEN t.direction = 'out' THEN t.amount_usd END) AS out_24h
FROM   transfers_clustered t
JOIN   cross_chain.clusters cp
       ON t.counterparty_cluster = cp.cluster_id
WHERE  t.asset = 'USDM'
  AND  cp.category IN ('bridge', 'dex')
  AND  t.block_timestamp >= now() - INTERVAL 24 HOUR
GROUP  BY venue, venue_type
ORDER  BY (in_24h + out_24h) DESC;
Data Solutions: bridge and DEX contracts are pre-attributed, so flows in/out of Wormhole, CCTP, Curve and Uniswap are categorized automatically. The issuer sees exactly how much supply is leaving the native chain and what it's being swapped into, no manual contract tagging.

-- Top USDM holders with entity attribution
SELECT c.entity_name AS holder,
       c.category,
       b.balance_usd,
       b.balance_usd / s.supply AS pct_supply
FROM   cross_chain.cluster_balances b
JOIN   cross_chain.clusters c ON b.cluster_id = c.cluster_id
CROSS  JOIN (SELECT SUM(balance_usd) AS supply
            FROM cross_chain.cluster_balances
            WHERE asset = 'USDM') s
WHERE  b.asset = 'USDM'
ORDER  BY b.balance_usd DESC
LIMIT  20;
